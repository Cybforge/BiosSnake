; =============================================================================
; Bare-Metal x86 BIOS ROM (64 KB) für QEMU – BLUE BOX EDITION
; Startet bei 0xFFFFFFF0 (0xF000:0xFFF0), initialisiert CAR/MTRR, i440FX PAM/DRB,
; PCI-VGA, Real-Mode-IVT, PIC, A20, 16550 UART, VGA-Font & zeichnet die Box!
; =============================================================================

[BITS 16]
[ORG 0x0000]                ; Gemappt auf Physisch 0xF0000 - 0xFFFFF (CS = 0xF000)

; --- RAM-Adressen fuer DVD-Logo Animation (im freien RAM bei 0x0000) ---
CLEAN_BUF equ 0x1000        ; 4000 Bytes Puffer (0x1000..0x1FA0) fuer Original-Screen
POS_X1    equ 0x2000        ; Position & Richtung Ausrufezeichen 1
POS_Y1    equ 0x2002
DIR_X1    equ 0x2004
DIR_Y1    equ 0x2006
POS_X2    equ 0x2008        ; Position & Richtung Ausrufezeichen 2
POS_Y2    equ 0x200A
DIR_X2    equ 0x200C
DIR_Y2    equ 0x200E
SOUND_TIMER equ 0x2010     ; Timer fuer PC-Speaker Tonlaenge

bios_start:
    cli                     ; 1. Maskierbare Interrupts sofort sperren
    cld                     ; Direction-Flag löschen

    ; --- NMI (Non-Maskable Interrupt) über CMOS-Port 0x70 deaktivieren ---
    mov al, 0x80
    out 0x70, al
    in  al, 0x71

    ; =========================================================================
    ; SCHRITT 1: CPU Cache & Cache-as-RAM (CAR) sauber initialisieren
    ; =========================================================================
    mov eax, 1
    cpuid
    bt  edx, 12             ; MTRR Feature-Bit prüfen
    jnc .skip_mtrr

    mov eax, cr0
    or  eax, 0x40000000     ; CR0.CD = 1 (Cache Disable)
    and eax, 0xDFFFFFFF     ; CR0.NW = 0
    mov cr0, eax
    invd                    ; Internen Cache invalidieren

    mov ecx, 0x0250         ; IA32_MTRR_FIX64K_00000
    mov eax, 0x06060606     ; Write-Back (0x06)
    mov edx, 0x06060606
    wrmsr

    mov ecx, 0x02FF         ; IA32_MTRR_DEF_TYPE
    mov eax, 0x00000C06     ; Enable MTRRs + Fixed MTRRs + Default Write-Back
    xor edx, edx
    wrmsr

.skip_mtrr:
    mov eax, cr0
    and eax, 0x9FFFFFFF     ; L1/L2 Cache aktivieren (CD=0, NW=0)
    mov cr0, eax

    ; =========================================================================
    ; SCHRITT 2: Northbridge (i440FX 00:00.0) & PCI-VGA (00:02.0) aufwecken
    ; =========================================================================
    mov eax, 0x80000058     ; Bus 0, Dev 0, Func 0, Offset 0x58 (DRAMC & PAM0..2)
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    in  eax, dx
    and eax, 0x000000FF     ; PAM0 (AH) = 0x00 (BIOS-ROM bleibt lesbar)
    or  eax, 0x33330000     ; PAM1 & PAM2 = R/W RAM
    out dx, eax

    mov eax, 0x8000005C     ; PAM3..PAM6 (Offset 0x5C..0x5F) auf R/W RAM
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    mov eax, 0x33333333
    out dx, eax

    mov eax, 0x80000060     ; DRAM Row Boundary Register (DRB0..DRB3)
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    in  eax, dx
    test eax, eax
    jnz .ram_ready
    mov eax, 0x08080808     ; Fallback: 64 MB DRAM
    out dx, eax
.ram_ready:

    ; PCI-VGA-Grafikkarte (Bus 0, Dev 2, Func 0, Offset 0x04 Command Reg)
    mov eax, 0x80001004
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    in  eax, dx
    or  eax, 0x00000007     ; I/O + Memory + Bus Master aktivieren
    out dx, eax

    ; =========================================================================
    ; SCHRITT 3: Stack, Segmente & Real-Mode-IVT (Interrupt Vector Table)
    ; =========================================================================
    xor ax, ax
    mov ss, ax
    mov sp, 0x7C00
    mov es, ax

    mov ax, 0xF000
    mov ds, ax

    xor di, di
    mov cx, 256
.ivt_init_loop:
    mov ax, dummy_isr
    stosw
    mov ax, 0xF000
    stosw
    loop .ivt_init_loop

    ; =========================================================================
    ; SCHRITT 4: Fast-A20, 8259 PIC & 16550 UART (inkl. FIFO)
    ; =========================================================================
    in  al, 0x92
    or  al, 0x02
    and al, 0xFE
    out 0x92, al

    mov al, 0x11
    out 0x20, al
    out 0xA0, al
    mov al, 0x08
    out 0x21, al
    mov al, 0x70
    out 0xA1, al
    mov al, 0x04
    out 0x21, al
    mov al, 0x02
    out 0xA1, al
    mov al, 0x01
    out 0x21, al
    out 0xA1, al
    mov al, 0xFF            ; Alle IRQs maskieren
    out 0x21, al
    out 0xA1, al

    ; COM1 (0x3F8) auf 115200 Baud, 8N1
    xor al, al
    mov dx, 0x3F9
    out dx, al
    mov al, 0x80
    mov dx, 0x3FB
    out dx, al
    mov al, 0x01
    mov dx, 0x3F8
    out dx, al
    xor al, al
    mov dx, 0x3F9
    out dx, al
    mov al, 0x03
    mov dx, 0x3FB
    out dx, al
    mov al, 0xC7
    mov dx, 0x3FA
    out dx, al
    mov al, 0x03
    mov dx, 0x3FC
    out dx, al

    ; =========================================================================
    ; SCHRITT 5: Kompletter Bare-Metal VGA-Bringup (80x25 Text Mode)
    ; =========================================================================
    mov dx, 0x3C2
    mov al, 0x67
    out dx, al

    mov dx, 0x3C4
    mov ax, 0x0100          ; Sequencer Sync-Reset
    out dx, ax

    mov si, vga_seq_table
    mov cx, 4
    mov ah, 1
.seq_loop:
    mov dx, 0x3C4
    mov al, ah
    out dx, al
    inc dx
    lodsb
    out dx, al
    inc ah
    loop .seq_loop

    mov dx, 0x3C4
    mov ax, 0x0300          ; Sync-Reset beenden
    out dx, ax

    mov dx, 0x3D4
    mov ax, 0x0011          ; CRTC Schreibschutz aufheben
    out dx, ax

    mov si, vga_crtc_table
    mov cx, 25
    xor ah, ah
.crtc_loop:
    mov dx, 0x3D4
    mov al, ah
    out dx, al
    inc dx
    lodsb
    out dx, al
    inc ah
    loop .crtc_loop

    mov si, vga_gc_table
    mov cx, 9
    xor ah, ah
.gc_loop:
    mov dx, 0x3CE
    mov al, ah
    out dx, al
    inc dx
    lodsb
    out dx, al
    inc ah
    loop .gc_loop

    mov dx, 0x3DA
    in  al, dx              ; Flip-Flop zurücksetzen
    mov dx, 0x3C0
    xor ah, ah
.ac_pal_loop:
    mov al, ah
    out dx, al
    out dx, al              ; Identitäts-Mapping 0..15 -> DAC 0..15
    inc ah
    cmp ah, 16
    jb  .ac_pal_loop

    mov si, vga_ac_ext_table
    mov cx, 5
.ac_ext_loop:
    mov al, ah
    out dx, al
    lodsb
    out dx, al
    inc ah
    loop .ac_ext_loop

    mov dx, 0x3DA
    in  al, dx
    mov dx, 0x3C0
    mov al, 0x20            ; Video-Output freigeben (PAS=1)
    out dx, al

    ; PEL Mask Register (0x3C6) auf 0xFF
    mov dx, 0x3C6
    mov al, 0xFF
    out dx, al

    ; =========================================================================
    ; SCHRITT 5.8: VGA DAC Palette für Schwarz, Blau, Grün & Weiß laden
    ; =========================================================================
    mov dx, 0x3C8
    xor al, al
    out dx, al              ; Start bei DAC-Index 0
    mov dx, 0x3C9
    mov si, vga_dac_table
    mov cx, 16 * 3          ; 16 Farben * 3 Bytes (R, G, B)
.dac_load_loop:
    lodsb
    out dx, al
    loop .dac_load_loop

    ; =========================================================================
    ; SCHRITT 6: Bitmap-Font sauber in VGA Plane 2 (0xA0000) laden
    ; =========================================================================
    mov dx, 0x3C4
    mov ax, 0x0100
    out dx, ax
    mov ax, 0x0402          ; SR02 = Plane 2
    out dx, ax
    mov ax, 0x0604          ; SR04 = Sequential
    out dx, ax
    mov ax, 0x0300
    out dx, ax

    mov dx, 0x3CE
    mov ax, 0x0204          ; GR04 = Plane 2
    out dx, ax
    mov ax, 0x0005          ; GR05 = Write Mode 0
    out dx, ax
    mov ax, 0x0406          ; GR06 = 0xA0000
    out dx, ax

    mov ax, 0xA000
    mov es, ax

    xor di, di
    xor ax, ax
    mov cx, 16384
    rep stosw

    mov si, font_table
.font_char_loop:
    lodsb
    test al, al
    jz  .font_done
    xor ah, ah
    shl ax, 5               ; Offset = ASCII * 32
    mov di, ax
    mov cx, 8
.font_row_loop:
    lodsb
    stosb                   ; Scanline 2n
    stosb                   ; Scanline 2n+1 (8x16 Font)
    loop .font_row_loop
    jmp .font_char_loop

.font_done:
    ; Zurück in den Odd/Even Text-Mode bei 0xB8000 schalten
    mov dx, 0x3C4
    mov ax, 0x0100
    out dx, ax
    mov ax, 0x0302          ; SR02 = Plane 0 & 1
    out dx, ax
    mov ax, 0x0204          ; SR04 = Odd/Even
    out dx, ax
    mov ax, 0x0300
    out dx, ax

    mov dx, 0x3CE
    mov ax, 0x0004
    out dx, ax
    mov ax, 0x1005
    out dx, ax
    mov ax, 0x0E06          ; GR06 = 0xB8000 Textfenster
    out dx, ax

    ; =========================================================================
    ; SCHRITT 7: "Hello World from BIOS!" + Blaue Box in der Mitte zeichnen
    ; =========================================================================
    mov ax, 0xB800
    mov es, ax
    xor di, di
    mov ax, 0x0A20          ; Schwarzer Hintergrund, grüne Textfarbe
    mov cx, 80 * 25
    rep stosw               ; Kompletten Screen löschen

    ; --- 7.1 "Hello World from BIOS!" oben links (Zeile 0, Spalte 0) ---
    xor di, di
    mov si, msg_hello
    mov ah, 0x0A            ; Attribut 0x0A = Hellgrün auf Schwarz
.print_hello:
    lodsb
    test al, al
    jz  .hello_done
    stosw
    call serial_putc        ; Parallel auf COM1 & Port 0xE9 ausgeben
    jmp .print_hello

.hello_done:
    ; Hardware-Cursor direkt hinter "Hello World from BIOS!" parken
    mov bx, di
    shr bx, 1
    mov dx, 0x3D4
    mov al, 0x0E
    out dx, al
    inc dx
    mov al, bh
    out dx, al
    dec dx
    mov al, 0x0F
    out dx, al
    inc dx
    mov al, bl
    out dx, al

    ; --- 7.2 Blaue Box in der Bildschirmmitte (Zeile 9..15, Spalte 24..55) ---
    ; Start-Offset für Zeile 9, Spalte 24: (9 * 80 + 24) * 2 = 1488
    mov di, (9 * 80 + 24) * 2
    mov dx, 7               ; Höhe der Box: 7 Zeilen (Zeile 9 bis 15)
.box_row_loop:
    push di
    mov cx, 32              ; Breite der Box: 32 Spalten (Spalte 24 bis 55)
    mov ax, 0x1F20          ; Attribut 0x1F (Weiß auf Blau), Zeichen ' '
    rep stosw
    pop di
    add di, 80 * 2          ; Nächste Bildschirmzeile (+160 Bytes)
    dec dx
    jnz .box_row_loop

    ; --- 7.3 Dezenter weißer Rahmen in die blaue Box zeichnen ---
    ; Obere Rahmenlinie (Zeile 9, Spalte 24..55)
    mov di, (9 * 80 + 24) * 2
    mov ax, 0x1F01          ; 0x01 = Ecke oben links '┌'
    stosw
    mov cx, 30
    mov ax, 0x1F05          ; 0x05 = Horizontale Linie '─'
    rep stosw
    mov ax, 0x1F02          ; 0x02 = Ecke oben rechts '┐'
    stosw

    ; Untere Rahmenlinie (Zeile 15, Spalte 24..55)
    mov di, (15 * 80 + 24) * 2
    mov ax, 0x1F03          ; 0x03 = Ecke unten links '└'
    stosw
    mov cx, 30
    mov ax, 0x1F05          ; 0x05 = Horizontale Linie '─'
    rep stosw
    mov ax, 0x1F04          ; 0x04 = Ecke unten rechts '┘'
    stosw

    ; Seitliche Rahmenlinien für Zeilen 10..14
    mov di, (10 * 80 + 24) * 2
    mov cx, 5
.side_border_loop:
    mov word [es:di], 0x1F06            ; Links '│'
    mov word [es:di + 31 * 2], 0x1F06   ; Rechts '│'
    add di, 80 * 2
    loop .side_border_loop

    ; --- 7.4 Weißen Text "die box geht!" exakt zentriert in die Box setzen ---
    ; Zeile 12 (Mitte von 9..15), Spalte 33: (12 * 80 + 33) * 2 = 1986
    mov di, (12 * 80 + 33) * 2
    mov si, msg_box
    mov ah, 0x1F            ; Attribut 0x1F = Strahlend Weiß auf Blau!
.print_box_text:
    lodsb
    test al, al
    jz  .all_done
    stosw
    call serial_putc
    jmp .print_box_text

.all_done:
    mov al, 13
    call serial_putc
    mov al, 10
    call serial_putc


    ; =========================================================================
    ; SCHRITT 8: DVD-Logo Animation – 2 fliegende gelbe Ausrufezeichen
    ; =========================================================================
    ; 1. Den fertigen Bildschirm (inkl. Box & Text) als sauberes Backup
    ;    nach CLEAN_BUF (0x0000:0x1000) sichern (4000 Bytes).
    mov ax, 0x0000
    mov es, ax
    mov ax, 0xB800
    mov ds, ax
    xor si, si
    mov di, CLEAN_BUF
    mov cx, 80 * 25
    rep movsw

    ; 2. Segmente fuer die Animation einrichten:
    ;    DS = 0x0000 (RAM fuer Backup & Variablen)
    ;    ES = 0xB800 (VGA-Textspeicher)
    mov ax, 0x0000
    mov ds, ax
    mov ax, 0xB800
    mov es, ax

    ; 3. Startpositionen und Richtungen der beiden Ausrufezeichen initialisieren
    ;    Ausrufezeichen 1: Startet links oben (X=2, Y=2), fliegt nach rechts-unten (+1, +1)
    mov word [POS_X1], 2
    mov word [POS_Y1], 2
    mov word [DIR_X1], 1
    mov word [DIR_Y1], 1

    ;    Ausrufezeichen 2: Startet rechts unten (X=72, Y=20), fliegt nach links-oben (-1, -1)
    mov word [POS_X2], 72
    mov word [POS_Y2], 20
    mov word [DIR_X2], -1
    mov word [DIR_Y2], -1
    mov byte [SOUND_TIMER], 0       ; Speaker initial aus

.anim_loop:
    ; --- Schritt A: Alte Positionen mit Original-Hintergrund restaurieren ---
    mov ax, [POS_Y1]
    mov bx, [POS_X1]
    call calc_offset
    mov ax, [CLEAN_BUF + di]
    mov [es:di], ax

    mov ax, [POS_Y2]
    mov bx, [POS_X2]
    call calc_offset
    mov ax, [CLEAN_BUF + di]
    mov [es:di], ax

    ; --- Schritt B: Position von Ausrufezeichen 1 berechnen & an Waenden abprallen ---
    xor cx, cx                      ; CX zaehlt Wandkontakte (1 = Wand, 2 = Ecke!)
    mov ax, [POS_X1]
    add ax, [DIR_X1]
    cmp ax, 0
    jg .chk_r1
    mov ax, 0
    mov word [DIR_X1], 1
    inc cx
    jmp .set_x1
.chk_r1:
    cmp ax, 79
    jl .set_x1
    mov ax, 79
    mov word [DIR_X1], -1
    inc cx
.set_x1:
    mov [POS_X1], ax

    mov ax, [POS_Y1]
    add ax, [DIR_Y1]
    cmp ax, 0
    jg .chk_b1
    mov ax, 0
    mov word [DIR_Y1], 1
    inc cx
    jmp .set_y1
.chk_b1:
    cmp ax, 24
    jl .set_y1
    mov ax, 24
    mov word [DIR_Y1], -1
    inc cx
.set_y1:
    mov [POS_Y1], ax

    ; Sound fuer Ausrufezeichen 1
    test cx, cx
    jz .snd_chk2
    cmp cx, 2
    je .corner_snd1
    ; Standard-Wandabpraller: 750 Hz (Divisor 1590 = 0x0636), 2 Frames (~70ms)
    mov bx, 0x0636
    mov dl, 2
    call play_sound
    jmp .snd_chk2
.corner_snd1:
    ; Eckenabpraller: hoher Jubel-Ping 1400 Hz (Divisor 852 = 0x0354), 4 Frames (~140ms)
    mov bx, 0x0354
    mov dl, 4
    call play_sound

.snd_chk2:
    ; --- Schritt C: Position von Ausrufezeichen 2 berechnen & an Waenden abprallen ---
    xor cx, cx                      ; CX zaehlt Wandkontakte
    mov ax, [POS_X2]
    add ax, [DIR_X2]
    cmp ax, 0
    jg .chk_r2
    mov ax, 0
    mov word [DIR_X2], 1
    inc cx
    jmp .set_x2
.chk_r2:
    cmp ax, 79
    jl .set_x2
    mov ax, 79
    mov word [DIR_X2], -1
    inc cx
.set_x2:
    mov [POS_X2], ax

    mov ax, [POS_Y2]
    add ax, [DIR_Y2]
    cmp ax, 0
    jg .chk_b2
    mov ax, 0
    mov word [DIR_Y2], 1
    inc cx
    jmp .set_y2
.chk_b2:
    cmp ax, 24
    jl .set_y2
    mov ax, 24
    mov word [DIR_Y2], -1
    inc cx
.set_y2:
    mov [POS_Y2], ax

    ; Sound fuer Ausrufezeichen 2
    test cx, cx
    jz .snd_done_bounce
    cmp cx, 2
    je .corner_snd2
    ; Standard-Wandabpraller: 950 Hz (Divisor 1256 = 0x04E8), 2 Frames (~70ms)
    mov bx, 0x04E8
    mov dl, 2
    call play_sound
    jmp .snd_done_bounce
.corner_snd2:
    ; Eckenabpraller: hoher Ping 1600 Hz (Divisor 745 = 0x02E9), 4 Frames (~140ms)
    mov bx, 0x02E9
    mov dl, 4
    call play_sound
.snd_done_bounce:

    ; --- Schritt D: Gelbes Ausrufezeichen 1 zeichnen ---
    mov ax, [POS_Y1]
    mov bx, [POS_X1]
    call calc_offset
    mov ah, [CLEAN_BUF + di + 1]    ; Original-Hintergrundfarbe (Schwarz oder Blau)
    and ah, 0xF0                    ; Behalte Hintergrund
    or  ah, 0x0E                    ; Strahlendes Gelb (0x0E) fuer Vordergrund
    mov al, '!'
    mov [es:di], ax

    ; --- Schritt E: Gelbes Ausrufezeichen 2 zeichnen ---
    mov ax, [POS_Y2]
    mov bx, [POS_X2]
    call calc_offset
    mov ah, [CLEAN_BUF + di + 1]    ; Original-Hintergrundfarbe
    and ah, 0xF0                    ; Behalte Hintergrund
    or  ah, 0x0E                    ; Strahlendes Gelb (0x0E) fuer Vordergrund
    mov al, '!'
    mov [es:di], ax

    ; --- Schritt F: Frame-Pause fuer fluessige Animation (~25 FPS) ---
    call delay_frame

    ; --- Schritt G: PC-Speaker Sound-Timer dekrementieren & ggf. stoppen ---
    cmp byte [SOUND_TIMER], 0
    jz .sound_off_done
    dec byte [SOUND_TIMER]
    jnz .sound_off_done
    ; Timer abgelaufen: Speaker stummschalten
    in  al, 0x61
    and al, 0xFC
    out 0x61, al
.sound_off_done:

    jmp .anim_loop

; Offset-Berechnung: DI = (AX * 80 + BX) * 2

; --- PC-Speaker Sound-Routine (PIT Kanal 2 / Port 0x61) ---
; Eingang: BX = Divisor (Frequenz = 1193182 / BX), DL = Dauer in Frames
play_sound:
    push ax
    ; PIT Kanal 2 auf Modus 3 (Rechteckwelle) konfigurieren
    mov al, 0xB6
    out 0x43, al
    mov al, bl
    out 0x42, al            ; Divisor LSB
    mov al, bh
    out 0x42, al            ; Divisor MSB

    ; Speaker anschalten (Bits 0 & 1 auf Port 0x61 setzen)
    in  al, 0x61
    or  al, 0x03
    out 0x61, al

    ; Timer setzen
    mov byte [SOUND_TIMER], dl
    pop ax
    ret

calc_offset:
    mov di, ax
    shl di, 4
    shl ax, 6
    add di, ax
    add di, bx
    shl di, 1
    ret

; Verzoegerung: Wartet ca. 35-40 Millisekunden (~25 FPS)
delay_frame:
    push eax
    push ebx
    rdtsc
    mov ebx, eax
.wait:
    rdtsc
    sub eax, ebx
    cmp eax, 35000000
    jb .wait
    pop ebx
    pop eax
    ret


; --- Hilfsroutine: Zeichen in AL über UART & QEMU-Debug-Port senden ---
serial_putc:
    push dx
    push bx
    mov bl, al
.wait:
    mov dx, 0x3FD
    in  al, dx
    test al, 0x20
    jz  .wait
    mov al, bl
    mov dx, 0x3F8
    out dx, al
    out 0xE9, al
    pop bx
    pop dx
    ret

dummy_isr:
    iret

; =============================================================================
; KONSTANTEN, VGA-REGISTER-TABELLEN, PALETTE & BITMAP-FONT
; =============================================================================

msg_hello:
    db "Hello World from BIOS!", 0

msg_box:
    db "die box geht!", 0

; Sequencer Registers SR01 - SR04
vga_seq_table:
    db 0x01, 0x03, 0x00, 0x02

; CRTC Registers (CR00 - CR18)
vga_crtc_table:
    db 0x5F, 0x4F, 0x50, 0x82, 0x55, 0x81, 0xBF, 0x1F
    db 0x00, 0x4F, 0x0D, 0x0E, 0x00, 0x00, 0x00, 0x00
    db 0x9C, 0x8E, 0x8F, 0x28, 0x1F, 0x96, 0xB9, 0xA3
    db 0xFF

; Graphics Controller Registers (GR00 - GR08)
vga_gc_table:
    db 0x00, 0x00, 0x00, 0x00, 0x00, 0x10, 0x0E, 0x00, 0xFF

; Attribute Controller Registers AR10 - AR14
vga_ac_ext_table:
    db 0x04, 0x00, 0x0F, 0x00, 0x00

; 16-Farben VGA DAC Palette (Je 3 Bytes: R, G, B im Bereich 0..63)
vga_dac_table:
    db  0,  0,  0           ; 0x00: Schwarz (Screen-Hintergrund)
    db  0, 12, 48           ; 0x01: Kräftiges BIOS-Blau (Box-Hintergrund)
    db  0, 42,  0           ; 0x02: Dunkelgrün
    db  0, 42, 42           ; 0x03: Cyan
    db 42,  0,  0           ; 0x04: Rot
    db 42,  0, 42           ; 0x05: Magenta
    db 42, 21,  0           ; 0x06: Braun
    db 42, 42, 42           ; 0x07: Hellgrau
    db 21, 21, 21           ; 0x08: Dunkelgrau
    db 21, 21, 63           ; 0x09: Hellblau
    db 10, 63, 20           ; 0x0A: Matrix-Hellgrün ("Hello World from BIOS!")
    db 21, 63, 63           ; 0x0B: Hellcyan
    db 63, 21, 21           ; 0x0C: Hellrot
    db 63, 21, 63           ; 0x0D: Pink
    db 63, 63, 21           ; 0x0E: Gelb
    db 63, 63, 63           ; 0x0F: Reines Weiß ("die box geht!" + Rahmen)

; Kompakte 8x8 Bitmap-Font-Tabelle (wird beim Upload auf 8x16 verdoppelt)
