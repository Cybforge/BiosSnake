; =============================================================================
; Bare-Metal x86 BIOS ROM (64 KB) for QEMU - BLUE BOX EDITION
; Starts at 0xFFFFFFF0 (0xF000:0xFFF0), initializes CAR/MTRR, i440FX PAM/DRB,
; PCI-VGA, Real-Mode IVT, PIC, A20, 16550 UART, VGA Font & draws the box!
; =============================================================================

[BITS 16]
[ORG 0x0000]                ; Mapped to Physical 0xF0000 - 0xFFFFF (CS = 0xF000)

; --- RAM Addresses for DVD-Logo Animation (in free RAM at 0x0000) ---
CLEAN_BUF equ 0x1000        ; 4000-byte buffer (0x1000..0x1FA0) for original screen
POS_X1    equ 0x2000        ; Position & direction of exclamation mark 1
POS_Y1    equ 0x2002
DIR_X1    equ 0x2004
DIR_Y1    equ 0x2006
POS_X2    equ 0x2008        ; Position & direction of exclamation mark 2
POS_Y2    equ 0x200A
DIR_X2    equ 0x200C
DIR_Y2    equ 0x200E
SOUND_TIMER equ 0x2010     ; Timer for PC Speaker tone duration

bios_start:
    cli                     ; 1. Disable maskable interrupts immediately
    cld                     ; Clear direction flag

    ; --- Disable NMI (Non-Maskable Interrupt) via CMOS Port 0x70 ---
    mov al, 0x80
    out 0x70, al
    in  al, 0x71

    ; =========================================================================
    ; STEP 1: Initialize CPU Cache & Cache-as-RAM (CAR) cleanly
    ; =========================================================================
    mov eax, 1
    cpuid
    bt  edx, 12             ; Check MTRR feature bit
    jnc .skip_mtrr

    mov eax, cr0
    or  eax, 0x40000000     ; CR0.CD = 1 (Cache Disable)
    and eax, 0xDFFFFFFF     ; CR0.NW = 0
    mov cr0, eax
    invd                    ; Invalidate internal cache

    mov ecx, 0x0250         ; IA32_MTRR_FIX64K_00000
    mov eax, 0x06060606     ; Write-Back (0x06)
    mov edx, 0x06060606
    wrmsr

    mov ecx, 0x02FF         ; IA32_MTRR_DEF_TYPE
    mov eax, 0x00000C06     ; Enable MTRRs + Fixed MTRRs + Default Write-Back
    xor edx, edx
    wrmsr

.skip_mtrr:
    mov eax, cr0
    and eax, 0x9FFFFFFF     ; Enable L1/L2 cache (CD=0, NW=0)
    mov cr0, eax

    ; =========================================================================
    ; STEP 2: Wake up Northbridge (i440FX 00:00.0) & PCI-VGA (00:02.0)
    ; =========================================================================
    mov eax, 0x80000058     ; Bus 0, Dev 0, Func 0, Offset 0x58 (DRAMC & PAM0..2)
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    in  eax, dx
    and eax, 0x000000FF     ; PAM0 (AH) = 0x00 (BIOS ROM remains readable)
    or  eax, 0x33330000     ; PAM1 & PAM2 = R/W RAM
    out dx, eax

    mov eax, 0x8000005C     ; Set PAM3..PAM6 (Offset 0x5C..0x5F) to R/W RAM
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    mov eax, 0x33333333
    out dx, eax

    mov eax, 0x80000060     ; DRAM Row Boundary Register (DRB0..DRB3)
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    in  eax, dx
    test eax, eax
    jnz .ram_ready
    mov eax, 0x08080808     ; Fallback: 64 MB DRAM
    out dx, eax
.ram_ready:

    ; PCI-VGA graphics card (Bus 0, Dev 2, Func 0, Offset 0x04 Command Reg)
    mov eax, 0x80001004
    mov dx, 0x0CF8
    out dx, eax
    mov dx, 0x0CFC
    in  eax, dx
    or  eax, 0x00000007     ; Enable I/O + Memory + Bus Master
    out dx, eax

    ; =========================================================================
    ; STEP 3: Stack, Segments & Real-Mode IVT (Interrupt Vector Table)
    ; =========================================================================
    xor ax, ax
    mov ss, ax
    mov sp, 0x7C00
    mov es, ax

    mov ax, 0xF000
    mov ds, ax

    xor di, di
    mov cx, 256
.ivt_init_loop:
    mov ax, dummy_isr
    stosw
    mov ax, 0xF000
    stosw
    loop .ivt_init_loop

    ; =========================================================================
    ; STEP 4: Fast-A20, 8259 PIC & 16550 UART (incl. FIFO)
    ; =========================================================================
    in  al, 0x92
    or  al, 0x02
    and al, 0xFE
    out 0x92, al

    mov al, 0x11
    out 0x20, al
    out 0xA0, al
    mov al, 0x08
    out 0x21, al
    mov al, 0x70
    out 0xA1, al
    mov al, 0x04
    out 0x21, al
    mov al, 0x02
    out 0xA1, al
    mov al, 0x01
    out 0x21, al
    out 0xA1, al
    mov al, 0xFF            ; Mask all IRQs
    out 0x21, al
    out 0xA1, al

    ; COM1 (0x3F8) at 115200 Baud, 8N1
    xor al, al
    mov dx, 0x3F9
    out dx, al
    mov al, 0x80
    mov dx, 0x3FB
    out dx, al
    mov al, 0x01
    mov dx, 0x3F8
    out dx, al
    xor al, al
    mov dx, 0x3F9
    out dx, al
    mov al, 0x03
    mov dx, 0x3FB
    out dx, al
    mov al, 0xC7
    mov dx, 0x3FA
    out dx, al
    mov al, 0x03
    mov dx, 0x3FC
    out dx, al

    ; =========================================================================
    ; STEP 5: Complete Bare-Metal VGA Bringup (80x25 Text Mode)
    ; =========================================================================
    mov dx, 0x3C2
    mov al, 0x67
    out dx, al

    mov dx, 0x3C4
    mov ax, 0x0100          ; Sequencer Sync Reset
    out dx, ax

    mov si, vga_seq_table
    mov cx, 4
    mov ah, 1
.seq_loop:
    mov dx, 0x3C4
    mov al, ah
    out dx, al
    inc dx
    lodsb
    out dx, al
    inc ah
    loop .seq_loop

    mov dx, 0x3C4
    mov ax, 0x0300          ; End Sync Reset
    out dx, ax

    mov dx, 0x3D4
    mov ax, 0x0011          ; Unlock CRTC write protection
    out dx, ax

    mov si, vga_crtc_table
    mov cx, 25
    xor ah, ah
.crtc_loop:
    mov dx, 0x3D4
    mov al, ah
    out dx, al
    inc dx
    lodsb
    out dx, al
    inc ah
    loop .crtc_loop

    mov si, vga_gc_table
    mov cx, 9
    xor ah, ah
.gc_loop:
    mov dx, 0x3CE
    mov al, ah
    out dx, al
    inc dx
    lodsb
    out dx, al
    inc ah
    loop .gc_loop

    mov dx, 0x3DA
    in  al, dx              ; Reset flip-flop
    mov dx, 0x3C0
    xor ah, ah
.ac_pal_loop:
    mov al, ah
    out dx, al
    out dx, al              ; Identity mapping 0..15 -> DAC 0..15
    inc ah
    cmp ah, 16
    jb  .ac_pal_loop

    mov si, vga_ac_ext_table
    mov cx, 5
.ac_ext_loop:
    mov al, ah
    out dx, al
    lodsb
    out dx, al
    inc ah
    loop .ac_ext_loop

    mov dx, 0x3DA
    in  al, dx
    mov dx, 0x3C0
    mov al, 0x20            ; Enable video output (PAS=1)
    out dx, al

    ; Set PEL Mask Register (0x3C6) to 0xFF
    mov dx, 0x3C6
    mov al, 0xFF
    out dx, al

    ; =========================================================================
    ; STEP 5.8: Load VGA DAC Palette for Black, Blue, Green & White
    ; =========================================================================
    mov dx, 0x3C8
    xor al, al
    out dx, al              ; Start at DAC index 0
    mov dx, 0x3C9
    mov si, vga_dac_table
    mov cx, 16 * 3          ; 16 colors * 3 bytes (R, G, B)
.dac_load_loop:
    lodsb
    out dx, al
    loop .dac_load_loop

    ; =========================================================================
    ; STEP 6: Load bitmap font cleanly into VGA Plane 2 (0xA0000)
    ; =========================================================================
    mov dx, 0x3C4
    mov ax, 0x0100
    out dx, ax
    mov ax, 0x0402          ; SR02 = Plane 2
    out dx, ax
    mov ax, 0x0604          ; SR04 = Sequential
    out dx, ax
    mov ax, 0x0300
    out dx, ax

    mov dx, 0x3CE
    mov ax, 0x0204          ; GR04 = Plane 2
    out dx, ax
    mov ax, 0x0005          ; GR05 = Write Mode 0
    out dx, ax
    mov ax, 0x0406          ; GR06 = 0xA0000
    out dx, ax

    mov ax, 0xA000
    mov es, ax

    xor di, di
    xor ax, ax
    mov cx, 16384
    rep stosw

    mov si, font_table
.font_char_loop:
    lodsb
    test al, al
    jz  .font_done
    xor ah, ah
    shl ax, 5               ; Offset = ASCII * 32
    mov di, ax
    mov cx, 8
.font_row_loop:
    lodsb
    stosb                   ; Scanline 2n
    stosb                   ; Scanline 2n+1 (8x16 Font)
    loop .font_row_loop
    jmp .font_char_loop

.font_done:
    ; Switch back to Odd/Even Text Mode at 0xB8000
    mov dx, 0x3C4
    mov ax, 0x0100
    out dx, ax
    mov ax, 0x0302          ; SR02 = Plane 0 & 1
    out dx, ax
    mov ax, 0x0204          ; SR04 = Odd/Even
    out dx, ax
    mov ax, 0x0300
    out dx, ax

    mov dx, 0x3CE
    mov ax, 0x0004
    out dx, ax
    mov ax, 0x1005
    out dx, ax
    mov ax, 0x0E06          ; GR06 = 0xB8000 text window
    out dx, ax

    ; =========================================================================
    ; STEP 7: Draw "Hello World from BIOS!" + Blue Box in the center
    ; =========================================================================
    mov ax, 0xB800
    mov es, ax
    xor di, di
    mov ax, 0x0A20          ; Black background, green text color
    mov cx, 80 * 25
    rep stosw               ; Clear entire screen

    ; --- 7.1 "Hello World from BIOS!" top-left (Row 0, Col 0) ---
    xor di, di
    mov si, msg_hello
    mov ah, 0x0A            ; Attribute 0x0A = Bright Green on Black
.print_hello:
    lodsb
    test al, al
    jz  .hello_done
    stosw
    call serial_putc        ; Output in parallel to COM1 & Port 0xE9
    jmp .print_hello

.hello_done:
    ; Park hardware cursor right after "Hello World from BIOS!"
    mov bx, di
    shr bx, 1
    mov dx, 0x3D4
    mov al, 0x0E
    out dx, al
    inc dx
    mov al, bh
    out dx, al
    dec dx
    mov al, 0x0F
    out dx, al
    inc dx
    mov al, bl
    out dx, al

    ; --- 7.2 Blue Box in screen center (Row 9..15, Col 24..55) ---
    ; Start offset for Row 9, Col 24: (9 * 80 + 24) * 2 = 1488
    mov di, (9 * 80 + 24) * 2
    mov dx, 7               ; Box height: 7 rows (Row 9 to 15)
.box_row_loop:
    push di
    mov cx, 32              ; Box width: 32 columns (Col 24 to 55)
    mov ax, 0x1F20          ; Attribute 0x1F (White on Blue), char ' '
    rep stosw
    pop di
    add di, 80 * 2          ; Next screen row (+160 bytes)
    dec dx
    jnz .box_row_loop

    ; --- 7.3 Draw subtle white border inside the blue box ---
    ; Top border line (Row 9, Col 24..55)
    mov di, (9 * 80 + 24) * 2
    mov ax, 0x1F01          ; 0x01 = Top-left corner '┌'
    stosw
    mov cx, 30
    mov ax, 0x1F05          ; 0x05 = Horizontal line '─'
    rep stosw
    mov ax, 0x1F02          ; 0x02 = Top-right corner '┐'
    stosw

    ; Bottom border line (Row 15, Col 24..55)
    mov di, (15 * 80 + 24) * 2
    mov ax, 0x1F03          ; 0x03 = Bottom-left corner '└'
    stosw
    mov cx, 30
    mov ax, 0x1F05          ; 0x05 = Horizontal line '─'
    rep stosw
    mov ax, 0x1F04          ; 0x04 = Bottom-right corner '┘'
    stosw

    ; Side border lines for Rows 10..14
    mov di, (10 * 80 + 24) * 2
    mov cx, 5
.side_border_loop:
    mov word [es:di], 0x1F06            ; Left '│'
    mov word [es:di + 31 * 2], 0x1F06   ; Right '│'
    add di, 80 * 2
    loop .side_border_loop

    ; --- 7.4 Place white text centered inside the box ---
    ; Row 12 (center of 9..15), Col 33: (12 * 80 + 33) * 2 = 1986
    mov di, (12 * 80 + 33) * 2
    mov si, msg_box
    mov ah, 0x1F            ; Attribute 0x1F = Bright White on Blue!
.print_box_text:
    lodsb
    test al, al
    jz  .all_done
    stosw
    call serial_putc
    jmp .print_box_text

.all_done:
    mov al, 13
    call serial_putc
    mov al, 10
    call serial_putc


    ; =========================================================================
    ; STEP 8: DVD-Logo Animation - 2 flying yellow exclamation marks
    ; =========================================================================
    ; 1. Save the rendered screen (incl. box & text) as a clean backup
    ;    to CLEAN_BUF (0x0000:0x1000) (4000 bytes).
    mov ax, 0x0000
    mov es, ax
    mov ax, 0xB800
    mov ds, ax
    xor si, si
    mov di, CLEAN_BUF
    mov cx, 80 * 25
    rep movsw

    ; 2. Set up segments for animation:
    ;    DS = 0x0000 (RAM for backup & variables)
    ;    ES = 0xB800 (VGA text memory)
    mov ax, 0x0000
    mov ds, ax
    mov ax, 0xB800
    mov es, ax

    ; 3. Initialize starting positions and directions of both exclamation marks
    ;    Exclamation mark 1: Starts top-left (X=2, Y=2), moves bottom-right (+1, +1)
    mov word [POS_X1], 2
    mov word [POS_Y1], 2
    mov word [DIR_X1], 1
    mov word [DIR_Y1], 1

    ;    Exclamation mark 2: Starts bottom-right (X=72, Y=20), moves top-left (-1, -1)
    mov word [POS_X2], 72
    mov word [POS_Y2], 20
    mov word [DIR_X2], -1
    mov word [DIR_Y2], -1
    mov byte [SOUND_TIMER], 0       ; Speaker initially off

.anim_loop:
    ; --- Step A: Restore old positions with original background ---
    mov ax, [POS_Y1]
    mov bx, [POS_X1]
    call calc_offset
    mov ax, [CLEAN_BUF + di]
    mov [es:di], ax

    mov ax, [POS_Y2]
    mov bx, [POS_X2]
    call calc_offset
    mov ax, [CLEAN_BUF + di]
    mov [es:di], ax

    ; --- Step B: Calculate position of exclamation mark 1 & bounce off walls ---
    xor cx, cx                      ; CX counts wall hits (1 = wall, 2 = corner!)
    mov ax, [POS_X1]
    add ax, [DIR_X1]
    cmp ax, 0
    jg .chk_r1
    mov ax, 0
    mov word [DIR_X1], 1
    inc cx
    jmp .set_x1
.chk_r1:
    cmp ax, 79
    jl .set_x1
    mov ax, 79
    mov word [DIR_X1], -1
    inc cx
.set_x1:
    mov [POS_X1], ax

    mov ax, [POS_Y1]
    add ax, [DIR_Y1]
    cmp ax, 0
    jg .chk_b1
    mov ax, 0
    mov word [DIR_Y1], 1
    inc cx
    jmp .set_y1
.chk_b1:
    cmp ax, 24
    jl .set_y1
    mov ax, 24
    mov word [DIR_Y1], -1
    inc cx
.set_y1:
    mov [POS_Y1], ax

    ; Sound for exclamation mark 1
    test cx, cx
    jz .snd_chk2
    cmp cx, 2
    je .corner_snd1
    ; Standard wall bounce: 750 Hz (Divisor 1590 = 0x0636), 2 frames (~70ms)
    mov bx, 0x0636
    mov dl, 2
    call play_sound
    jmp .snd_chk2
.corner_snd1:
    ; Corner bounce: high chime ping 1400 Hz (Divisor 852 = 0x0354), 4 frames (~140ms)
    mov bx, 0x0354
    mov dl, 4
    call play_sound

.snd_chk2:
    ; --- Step C: Calculate position of exclamation mark 2 & bounce off walls ---
    xor cx, cx                      ; CX counts wall hits
    mov ax, [POS_X2]
    add ax, [DIR_X2]
    cmp ax, 0
    jg .chk_r2
    mov ax, 0
    mov word [DIR_X2], 1
    inc cx
    jmp .set_x2
.chk_r2:
    cmp ax, 79
    jl .set_x2
    mov ax, 79
    mov word [DIR_X2], -1
    inc cx
.set_x2:
    mov [POS_X2], ax

    mov ax, [POS_Y2]
    add ax, [DIR_Y2]
    cmp ax, 0
    jg .chk_b2
    mov ax, 0
    mov word [DIR_Y2], 1
    inc cx
    jmp .set_y2
.chk_b2:
    cmp ax, 24
    jl .set_y2
    mov ax, 24
    mov word [DIR_Y2], -1
    inc cx
.set_y2:
    mov [POS_Y2], ax

    ; Sound for exclamation mark 2
    test cx, cx
    jz .snd_done_bounce
    cmp cx, 2
    je .corner_snd2
    ; Standard wall bounce: 950 Hz (Divisor 1256 = 0x04E8), 2 frames (~70ms)
    mov bx, 0x04E8
    mov dl, 2
    call play_sound
    jmp .snd_done_bounce
.corner_snd2:
    ; Corner bounce: high ping 1600 Hz (Divisor 745 = 0x02E9), 4 frames (~140ms)
    mov bx, 0x02E9
    mov dl, 4
    call play_sound
.snd_done_bounce:

    ; --- Step D: Draw yellow exclamation mark 1 ---
    mov ax, [POS_Y1]
    mov bx, [POS_X1]
    call calc_offset
    mov ah, [CLEAN_BUF + di + 1]    ; Original background color (Black or Blue)
    and ah, 0xF0                    ; Keep background
    or  ah, 0x0E                    ; Bright Yellow (0x0E) for foreground
    mov al, '!'
    mov [es:di], ax

    ; --- Step E: Draw yellow exclamation mark 2 ---
    mov ax, [POS_Y2]
    mov bx, [POS_X2]
    call calc_offset
    mov ah, [CLEAN_BUF + di + 1]    ; Original background color
    and ah, 0xF0                    ; Keep background
    or  ah, 0x0E                    ; Bright Yellow (0x0E) for foreground
    mov al, '!'
    mov [es:di], ax

    ; --- Step F: Frame delay for smooth animation (~25 FPS) ---
    call delay_frame

    ; --- Step G: Decrement PC Speaker sound timer & stop if expired ---
    cmp byte [SOUND_TIMER], 0
    jz .sound_off_done
    dec byte [SOUND_TIMER]
    jnz .sound_off_done
    ; Timer expired: mute speaker
    in  al, 0x61
    and al, 0xFC
    out 0x61, al
.sound_off_done:

    jmp .anim_loop

; Offset calculation: DI = (AX * 80 + BX) * 2

; --- PC Speaker Sound Routine (PIT Channel 2 / Port 0x61) ---
; Input: BX = Divisor (Frequency = 1193182 / BX), DL = Duration in frames
play_sound:
    push ax
    ; Configure PIT Channel 2 to Mode 3 (square wave)
    mov al, 0xB6
    out 0x43, al
    mov al, bl
    out 0x42, al            ; Divisor LSB
    mov al, bh
    out 0x42, al            ; Divisor MSB

    ; Turn speaker on (set bits 0 & 1 on Port 0x61)
    in  al, 0x61
    or  al, 0x03
    out 0x61, al

    ; Set timer
    mov byte [SOUND_TIMER], dl
    pop ax
    ret

calc_offset:
    mov di, ax
    shl di, 4
    shl ax, 6
    add di, ax
    add di, bx
    shl di, 1
    ret

; Delay: Waits approx. 35-40 milliseconds (~25 FPS)
delay_frame:
    push eax
    push ebx
    rdtsc
    mov ebx, eax
.wait:
    rdtsc
    sub eax, ebx
    cmp eax, 35000000
    jb .wait
    pop ebx
    pop eax
    ret


; --- Helper routine: Send character in AL via UART & QEMU debug port ---
serial_putc:
    push dx
    push bx
    mov bl, al
.wait:
    mov dx, 0x3FD
    in  al, dx
    test al, 0x20
    jz  .wait
    mov al, bl
    mov dx, 0x3F8
    out dx, al
    out 0xE9, al
    pop bx
    pop dx
    ret

dummy_isr:
    iret

; =============================================================================
; CONSTANTS, VGA REGISTER TABLES, PALETTE & BITMAP FONT
; =============================================================================

msg_hello:
    db "Hello World from BIOS!", 0

msg_box:
    db "Box is working!", 0

; Sequencer Registers SR01 - SR04
vga_seq_table:
    db 0x01, 0x03, 0x00, 0x02

; CRTC Registers (CR00 - CR18)
vga_crtc_table:
    db 0x5F, 0x4F, 0x50, 0x82, 0x55, 0x81, 0xBF, 0x1F
    db 0x00, 0x4F, 0x0D, 0x0E, 0x00, 0x00, 0x00, 0x00
    db 0x9C, 0x8E, 0x8F, 0x28, 0x1F, 0x96, 0xB9, 0xA3
    db 0xFF

; Graphics Controller Registers (GR00 - GR08)
vga_gc_table:
    db 0x00, 0x00, 0x00, 0x00, 0x00, 0x10, 0x0E, 0x00, 0xFF

; Attribute Controller Registers AR10 - AR14
vga_ac_ext_table:
    db 0x04, 0x00, 0x0F, 0x00, 0x00

; 16-Color VGA DAC Palette (3 bytes each: R, G, B in range 0..63)
vga_dac_table:
    db  0,  0,  0           ; 0x00: Black (Screen background)
    db  0, 12, 48           ; 0x01: Deep BIOS Blue (Box background)
    db  0, 42,  0           ; 0x02: Dark Green
    db  0, 42, 42           ; 0x03: Cyan
    db 42,  0,  0           ; 0x04: Red
    db 42,  0, 42           ; 0x05: Magenta
    db 42, 21,  0           ; 0x06: Brown
    db 42, 42, 42           ; 0x07: Light Gray
    db 21, 21, 21           ; 0x08: Dark Gray
    db 21, 21, 63           ; 0x09: Light Blue
    db 10, 63, 20           ; 0x0A: Matrix Bright Green ("Hello World from BIOS!")
    db 21, 63, 63           ; 0x0B: Light Cyan
    db 63, 21, 21           ; 0x0C: Light Red
    db 63, 21, 63           ; 0x0D: Pink
    db 63, 63, 21           ; 0x0E: Yellow
    db 63, 63, 63           ; 0x0F: Pure White (Box text + border)

; Compact 8x8 bitmap font table (doubled to 8x16 during upload)
font_table:
    ; --- Border glyphs in slots 0x01 to 0x06 ---
    db 0x01, 0x00, 0x00, 0x00, 0x1F, 0x18, 0x18, 0x18, 0x18 ; '┌'
    db 0x02, 0x00, 0x00, 0x00, 0xF8, 0x18, 0x18, 0x18, 0x18 ; '┐'
    db 0x03, 0x18, 0x18, 0x18, 0x1F, 0x00, 0x00, 0x00, 0x00 ; '└'
    db 0x04, 0x18, 0x18, 0x18, 0xF8, 0x00, 0x00, 0x00, 0x00 ; '┘'
    db 0x05, 0x00, 0x00, 0x00, 0xFF, 0x00, 0x00, 0x00, 0x00 ; '─'
    db 0x06, 0x18, 0x18, 0x18, 0x18, 0x18, 0x18, 0x18, 0x18 ; '│'

    ; --- Text glyphs ---
    db ' ',  0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
    db '!',  0x18, 0x3C, 0x3C, 0x18, 0x18, 0x00, 0x18, 0x00
    db 'B',  0xFC, 0x66, 0x66, 0x7C, 0x66, 0x66, 0xFC, 0x00
    db 'H',  0xC6, 0xC6, 0xC6, 0xFE, 0xC6, 0xC6, 0xC6, 0x00
